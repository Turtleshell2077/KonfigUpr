# Short tour of all commands (works with any VFS and without VFS)

# whoami and the service command vfs-info
whoami
vfs-info

# ls, cd, tac and rev work with the loaded VFS
ls
ls -l /
cd /
cd ..
tac readme.txt
rev readme.txt

# a comment after a command is ignored
ls /   # this part is ignored

# an unknown command is an error, the script goes on
foo bar

# exit ends the work
exit
